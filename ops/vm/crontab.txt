# The VM's whole schedule, versioned with the code. Installed (replacing the old one) with
#     crontab ~/midfreq/current/ops/vm/crontab.txt
# and checked by tests/e2e/test_vm_schedule.py. Times are the VM's clock, Asia/Singapore (SGT).
# Nothing here talks to GitHub: code arrives only as releases (ops/README.md).
#
# m  h      dom mon dow  command
SHELL=/bin/bash
# cron's default PATH lacks /snap/bin, where the VM's gcloud lives (sync_logs.sh)
PATH=/usr/local/bin:/usr/bin:/bin:/snap/bin

# S&P 100 forward test. Fires at 21:35 and 22:35 because the US open moves with daylight
# saving (09:30 ET = 21:30 SGT under EDT, 22:30 SGT under EST); the launcher only proceeds
# within 30 minutes after the open, so exactly one of the two runs. The trading node runs it
# (config/node_sp100.yaml, paper account); to go back to the session runner, comment the node's
# line and remove the leading "#ROLLBACK " from the next one (docs/NODE_RUNBOOK.md).
35 21,22 * * 1-5  MIDFREQ_ROOT=$HOME/midfreq/runtime NODE_CONFIG=config/node_sp100.yaml NODE_MODE=paper $HOME/midfreq/current/ops/run_node.sh --scheduled >> $HOME/midfreq/logs/cron_sp100.log 2>&1
#ROLLBACK 35 21,22 * * 1-5  MIDFREQ_ROOT=$HOME/midfreq/runtime $HOME/midfreq/current/ops/run_sp100_forward_test.sh --scheduled >> $HOME/midfreq/logs/cron_sp100.log 2>&1

# Hong Kong roster: paused while the S&P 100 test runs (the two exceed the account's
# 100-symbol subscription quota). To resume, remove the leading "#PAUSED ".
#PAUSED 25 9 * * 1-5  MIDFREQ_ROOT=$HOME/midfreq/runtime $HOME/midfreq/current/ops/run_daily_candidates.sh >> $HOME/midfreq/logs/cron_roster.log 2>&1

# The bot's record to the hosted dashboard's bucket, every 2 minutes (does nothing until the
# uploader key is installed; ops/README.md).
*/2 * * * *  $HOME/midfreq/current/ops/vm/sync_logs.sh >> $HOME/midfreq/logs/sync_logs.log 2>&1

# The broker accounts with `publish: true` (runtime/config/connections.yaml) as snapshots in
# runtime/live_sessions/broker_<id>.json, for the dashboards, every 5 minutes. Read-only; only
# failures are logged.
*/5 * * * *  $HOME/midfreq/current/ops/vm/broker_snapshot.sh >> $HOME/midfreq/logs/broker_snapshot.log 2>&1

# RAM, swap and the largest processes, once a minute (keeps ~10 days).
* * * * *  $HOME/midfreq/current/ops/vm/memlog.sh

# Daily maintenance at 17:05, when no session runs: monthly log archive, OpenD and moomoo SDK
# log retention, one health line.
5 17 * * *  $HOME/midfreq/current/ops/vm/maintain.sh >> $HOME/midfreq/logs/maintain.log 2>&1
